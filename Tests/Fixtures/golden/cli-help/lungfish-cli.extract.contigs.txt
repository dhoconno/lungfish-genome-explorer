OVERVIEW: Extract selected contigs from an assembly FASTA or managed assembly
result

USAGE: lungfish-cli extract contigs <options>

OPTIONS:
  --assembly <assembly>   Managed assembly output directory containing
                          assembly-result.json
  --contigs <contigs>     Contigs FASTA path
  --contig <contig>       Contig name to extract (repeatable)
  --contig-file <contig-file>
                          Text file with one contig name per line (repeatable)
  -o, --output <output>   Output FASTA path
  --bundle                Create a derived .lungfishref bundle in the project
  --bundle-name <bundle-name>
                          Bundle display name for --bundle mode
  --project-root <project-root>
                          Project root directory for --bundle mode
  --line-width <line-width>
                          FASTA line width (default: 60) (default: 60)
  --format <format>       Output format: text, json, tsv (default: text)
                          (values: text, json, tsv; default: text)
  -v, --verbose           Increase output verbosity (can be repeated: -v, -vv,
                          -vvv)
  -q, --quiet             Suppress non-essential output (errors only)
  --progress              Show progress bar (default: auto-detect TTY)
  --no-progress           Disable progress bar
  --debug                 Enable debug output with detailed logging
  --log-file <log-file>   Write detailed logs to file
  --no-color              Disable colored output
  -t, --threads <threads> Number of threads to use (default: auto)
  --version               Show the version.
  -h, --help              Show help information.
