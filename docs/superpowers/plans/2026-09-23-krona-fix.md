## Confirmed diagnosis

1. `SunburstGeometry.swift:95-121` builds paths with `90 - angleDegrees`, outer `clockwise: true`, inner `false`. In the flipped view, labels and polar hit tests use `(cx + r*sin(a), cy - r*cos(a))`. The paths instead occupy the vertical reflection `(cx + r*sin(a), cy + r*cos(a))`. Correct numeric path angles are `angleDegrees - 90`, outer `clockwise: false`, inner `true`. This is a path-coordinate issue; do not flip the whole view or change `polarCoordinates`.
2. `containsPoint` independently normalizes both endpoints. For a full turn, both become zero and only the top ray hits. It also calls a zero-length span a hit at its endpoint. Preserve unnormalized span information before existing wrapped-angle logic.
3. `TaxonomySunburstView.swift:399` rotates by `-rotation`, but its own documented tangent is `(cos(a), sin(a))`; the local CGContext rotation must use positive `rotation` in the flipped drawing coordinates. Keep the existing upright-text half-turn adjustment.
4. Layout invalidation clears segment geometry but leaves `hoveredNode` and its floating tooltip attached to the previous layout. Clear hover when invalidating; do not make selection semantics part of this repair.
5. `TaxonomyTooltipView.update` currently sets `percentOfClassified = node.fractionClade * 100`. `TaxonNode` explicitly defines that fraction against **total** reads, so this duplicates `% of total` under the wrong label. Pass `tree.classifiedReads` from the sunburst.

### Evidence gathered without changing product code

All four 90-degree wedges produced `path.contains(expectedMidpoint) == false`, `path.contains(verticallyReflectedMidpoint) == true`, and `containsPoint(expectedPolarMidpoint) == true`. Changing only the two angle conversions and two arc directions in the diagnostic copy reversed both path assertions to the expected result.

The full ring's path contains its midpoint, while polar containment at `[0, π/2, π, 3π/2]` returns `[true, false, false, false]`. A zero-span segment returns true at angle zero.

A balanced synthetic parent with clade 100, direct 10, and children 70 and 20 produces 252° and 72° sectors and a correct 36° gap. Outer spikes can legitimately result from uneven tree depth, terminal taxa, direct assignments, and tiny aggregated sectors; this repair must not erase that information.

Existing tests miss the defects because they test angle math independently of rendered paths, check only that paths are nonempty, and merely check that full-ring segments have 360° spans. import AppKit
import XCTest
@testable import LungfishApp

@MainActor
final class TaxonomySunburstRegressionTests: XCTestCase {
    private func tree(singleChild: Bool = false) -> TaxonTree {
        func node(_ id: Int, _ name: String, _ clade: Int, _ direct: Int) -> TaxonNode {
            TaxonNode(taxId: id, name: name, rank: id == 1 ? .root : .phylum,
                      depth: id == 1 ? 0 : 1, readsDirect: direct, readsClade: clade,
                      fractionClade: Double(clade) / 1000,
                      fractionDirect: Double(direct) / 1000,
                      parentTaxId: id == 1 ? nil : 1)
        }
        let root = node(1, "Root", 100, singleChild ? 0 : 10)
        root.children = singleChild
            ? [node(2, "Only taxon", 100, 100)]
            : [node(2, "Alpha", 70, 70), node(3, "Beta", 20, 20)]
        for child in root.children { child.parent = root }
        return TaxonTree(root: root, unclassifiedNode: nil, totalReads: 1000)
    }

    private func point(_ layout: SunburstLayout, radius: CGFloat, angle: CGFloat) -> NSPoint {
        NSPoint(x: layout.center.x + radius * sin(angle),
                y: layout.center.y - radius * cos(angle))
    }

    private func segment(_ start: CGFloat, _ end: CGFloat) -> SunburstSegment {
        SunburstSegment(node: tree().root, ring: 0, innerRadius: 40, outerRadius: 80,
                        startAngle: start, endAngle: end, color: .red, isOther: false)
    }

    func testQuarterPathsAgreeWithPolarCoordinates() {
        let layout = SunburstLayout(tree: tree(), bounds: CGRect(x: 0, y: 0, width: 400, height: 400))
        for quadrant in 0..<4 {
            let start = CGFloat(quadrant) * .pi / 2
            let s = segment(start, start + .pi / 2)
            let inside = point(layout, radius: 60, angle: s.midAngle)
            let reflected = NSPoint(x: inside.x, y: 2 * layout.center.y - inside.y)
            XCTAssertTrue(s.bezierPath(center: layout.center).contains(inside))
            XCTAssertFalse(s.bezierPath(center: layout.center).contains(reflected))
            XCTAssertTrue(s.containsPoint(radius: 60, angle: s.midAngle))
        }
    }

    func testFullCircleHitsEveryCardinalPoint() {
        let layout = SunburstLayout(tree: tree(singleChild: true),
                                   bounds: CGRect(x: 0, y: 0, width: 400, height: 400))
        let segments = layout.computeSegments()
        let s = segments[0]
        for angle in [CGFloat(0), .pi / 2, .pi, 3 * .pi / 2] {
            let p = point(layout, radius: s.midRadius, angle: angle)
            XCTAssertTrue(s.bezierPath(center: layout.center).contains(p))
            XCTAssertEqual(layout.hitTest(point: p, segments: segments)?.node.taxId, 2)
        }
        XCTAssertFalse(s.containsPoint(radius: s.innerRadius - 1, angle: .pi))
        XCTAssertFalse(s.containsPoint(radius: s.outerRadius + 1, angle: .pi))
    }

    func testSeamAndEmptySpan() {
        let wrapped = segment(7 * .pi / 4, 9 * .pi / 4)
        XCTAssertTrue(wrapped.containsPoint(radius: 60, angle: 0))
        XCTAssertTrue(wrapped.containsPoint(radius: 60, angle: 2 * .pi))
        XCTAssertFalse(wrapped.containsPoint(radius: 60, angle: .pi))
        XCTAssertFalse(segment(0, 0).containsPoint(radius: 60, angle: 0))
        // Also retain existing normalized-endpoint wrap semantics.
        XCTAssertTrue(segment(7 * .pi / 4, .pi / 4).containsPoint(radius: 60, angle: 0))
    }

    func testDirectReadGapAndOtherPreserveMass() {
        let t = tree()
        for threshold in [0.001, 0.25] {
            let layout = SunburstLayout(tree: t,
                bounds: CGRect(x: 0, y: 0, width: 400, height: 400),
                minFractionToShow: threshold)
            let segments = layout.computeSegments().filter { $0.ring == 0 }
            XCTAssertEqual(segments.count, 2)
            XCTAssertEqual(segments[0].angularSpanDegrees, 252, accuracy: 1e-8)
            XCTAssertEqual(segments[1].angularSpanDegrees, 72, accuracy: 1e-8)
            XCTAssertEqual(segments[1].isOther, threshold == 0.25)
            let gap = point(layout, radius: segments[0].midRadius, angle: 1.9 * .pi)
            XCTAssertNil(layout.hitTest(point: gap, segments: segments))
            XCTAssertTrue(segments.allSatisfy { !$0.bezierPath(center: layout.center).contains(gap) })
        }
        XCTAssertEqual(t.root.readsClade, 100)
        XCTAssertEqual(t.root.readsDirect, 10)
        XCTAssertEqual(t.totalReads, 1000)
    }
}

private func event(_ type: NSEvent.EventType, in view: NSView,
                   at point: NSPoint, clicks: Int = 1) throws -> NSEvent {
    try XCTUnwrap(NSEvent.mouseEvent(with: type,
        location: view.convert(point, to: nil), modifierFlags: [], timestamp: 0,
        windowNumber: view.window?.windowNumber ?? 0, context: nil,
        eventNumber: 0, clickCount: clicks, pressure: 1))
}

func testMouseCallbacksFollowRenderedSectorsAndResize() throws {
    let t = tree()
    let view = TaxonomySunburstView(frame: NSRect(x: 0, y: 0, width: 400, height: 400))
    view.tree = t
    var selected: TaxonNode?
    var rightClicked: TaxonNode?
    var emptyRightClicks = 0
    view.onNodeSelected = { selected = $0 }
    view.onNodeRightClicked = { node, _ in rightClicked = node }
    view.onEmptySpaceRightClicked = { _ in emptyRightClicks += 1 }
    for size in [NSSize(width: 400, height: 400), NSSize(width: 700, height: 500)] {
        view.setFrameSize(size)
        view.layoutSubtreeIfNeeded()
        let layout = SunburstLayout(tree: t, bounds: view.bounds)
        let sectors = layout.computeSegments()
        for s in sectors where !s.isOther {
            let p = point(layout, radius: s.midRadius, angle: s.midAngle)
            XCTAssertTrue(s.bezierPath(center: layout.center).contains(p))
            view.mouseMoved(with: try event(.mouseMoved, in: view, at: p))
            XCTAssertTrue(view.hoveredNode === s.node)
            view.mouseDown(with: try event(.leftMouseDown, in: view, at: p))
            XCTAssertTrue(view.selectedNode === s.node)
            XCTAssertTrue(selected === s.node)
            view.rightMouseDown(with: try event(.rightMouseDown, in: view, at: p))
            XCTAssertTrue(rightClicked === s.node)
        }
        let gap = point(layout, radius: sectors[0].midRadius, angle: 1.9 * .pi)
        view.mouseMoved(with: try event(.mouseMoved, in: view, at: gap))
        XCTAssertNil(view.hoveredNode)
        view.rightMouseDown(with: try event(.rightMouseDown, in: view, at: gap))
    }
    XCTAssertEqual(emptyRightClicks, 2)
    let layout = SunburstLayout(tree: t, bounds: view.bounds)
    let s = layout.computeSegments()[0]
    let p = point(layout, radius: s.midRadius, angle: s.midAngle)
    view.mouseMoved(with: try event(.mouseMoved, in: view, at: p))
    XCTAssertNotNil(view.hoveredNode)
    view.mouseDown(with: try event(.leftMouseDown, in: view, at: p, clicks: 2))
    XCTAssertTrue(view.centerNode === s.node)
    XCTAssertNil(view.hoveredNode)
    view.zoomToRoot()
    view.hoveredNode = t.root.children[0]
    view.tree = tree(singleChild: true)
    XCTAssertNil(view.hoveredNode)
}
